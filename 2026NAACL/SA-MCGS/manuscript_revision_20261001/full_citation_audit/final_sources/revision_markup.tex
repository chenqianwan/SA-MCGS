\DeclareRobustCommand{\revadd}[1]{%
  \ifrevisionmarks{\color{blue}#1}\else#1\fi}
\DeclareRobustCommand{\revdel}[1]{%
  \ifrevisionmarks{\color{black}\sout{#1}}\fi}
\DeclareRobustCommand{\revreplace}[2]{%
  \ifrevisionmarks\revdel{#1}\space\fi\revadd{#2}}
% September 25: retain earlier blue additions; this round is dark green.
\definecolor{revisiongreen}{HTML}{16803C}
\DeclareRobustCommand{\revnew}[1]{%
  \ifrevisionmarks{\color{revisiongreen}#1}\else#1\fi}
\DeclareRobustCommand{\revnewreplace}[2]{%
  \ifrevisionmarks\revdel{#1}\space\fi\revnew{#2}}
% October 1: M1/M3 changes are red; earlier blue/green additions keep their color.
\definecolor{revisionred}{HTML}{C00000}
\DeclareRobustCommand{\revred}[1]{%
  \ifrevisionmarks{\color{revisionred}#1}\else#1\fi}
\DeclareRobustCommand{\revredreplace}[2]{%
  \ifrevisionmarks\revdel{#1}\space\fi\revred{#2}}
% Keep retired black tables in the marked view with individually struck cells.
\newcommand{\revdeletedtable}[1]{\ifrevisionmarks#1\fi}
% Preserve the prior asset reference in source while displaying the updated plot.
\newcommand{\revasset}[2]{#2}
\newcommand{\revoldlabel}[1]{%
  \ifstrequal{#1}{tab:appendix_breakdowns}{\label{#1:previous}}{\label{#1}}}
\newif\ifrevisionblock
\newif\ifrevisionnewblock
\newif\ifrevisionredblock
\DeclareCaptionFont{revisionnew}{\ifrevisionmarks\color{revisiongreen}\fi}
\DeclareCaptionFont{revisionred}{\ifrevisionmarks\color{revisionred}\fi}
\newenvironment{revadded}{%
  \begingroup\revisionblocktrue\ifrevisionmarks\color{blue}\fi
}{\endgroup}
\newenvironment{revnewblock}{%
  \begingroup\revisionnewblocktrue\captionsetup{font+=revisionnew}%
  \ifrevisionmarks\color{revisiongreen}\fi
}{\endgroup}
\newenvironment{revredblock}{%
  \begingroup\revisionredblocktrue\captionsetup{font+=revisionred}%
  \ifrevisionmarks\color{revisionred}\fi
}{\endgroup}

% LaTeX resets colors when it opens a float. Restore revision blue only for
% floats constructed inside a newly added block; original floats stay intact.
\makeatletter
\apptocmd{\@floatboxreset}{%
  \ifrevisionredblock\ifrevisionmarks\color{revisionred}\fi
  \else\ifrevisionnewblock\ifrevisionmarks\color{revisiongreen}\fi
  \else\ifrevisionblock\ifrevisionmarks\color{blue}\fi\fi\fi\fi
}{}{\PackageError{revision-markup}{Could not color added floats}{Check the float reset hook.}}
\makeatother
